Quick Start 
# Clone repos side by side
git clone https://github.com/abhijitkrm/monadbft-go.git
git clone https://github.com/abhijitkrm/monad-evm.git

# Build and test
cd monadbft-go
go build ./... && go test ./...

# Run single node
cd ../monad-evm/evmd && go build -o evmd .
./evmd init mynode --chain-id monad-1
./evmd start --engine=monadbft

# Run 4-node devnet
cd ../../monadbft-go/bridge
go build -o devnet ./cmd/devnet
./devnet init-files -n 4 -o ./devnet-run -transport tcp
./devnet soak -o ./devnet-run -duration 3m -rate 200
